package com.android.bedstead.remoteframeworkclasses;

import android.app.admin.PackageIdentifier;
import android.os.Parcel;
import android.os.Parcelable;

import com.google.android.enterprise.connectedapps.annotations.CustomParcelableWrapper;
import com.google.android.enterprise.connectedapps.internal.Bundler;
import com.google.android.enterprise.connectedapps.internal.BundlerType;

/**
 * Custom Parcelable to transport a {@link PackageIdentifier}.
 */
@CustomParcelableWrapper(originalType = PackageIdentifier.class)
public final class ParcelablePackageIdentifier implements Parcelable {

    private final String mPackageName;

    /**
     * Create a wrapper for a given {@link PackageIdentifier}.
     */
    public static ParcelablePackageIdentifier of(
            Bundler bundler, BundlerType type,
            PackageIdentifier identifier) {

        return new ParcelablePackageIdentifier(identifier, bundler, type);
    }

    // Constructor to create the Parcelable from the original object
    private ParcelablePackageIdentifier(
            PackageIdentifier identifier,
            Bundler bundler,
            BundlerType type) {
        mPackageName = identifier.getPackageName();
    }

    // Constructor to recreate the Parcelable from a Parcel
    private ParcelablePackageIdentifier(Parcel in) {
        mPackageName = in.readString();
    }

    // Method to reconstruct the original PackageIdentifier object
    public PackageIdentifier get() {
        return new PackageIdentifier(mPackageName);
    }

    public String getPackageName() {
        return mPackageName;
    }

    @Override
    public void writeToParcel(Parcel dest, int flags) {
        dest.writeString(mPackageName);
    }

    @Override
    public int describeContents() {
        return 0;
    }

    public static final Creator<ParcelablePackageIdentifier> CREATOR =
        new Creator<ParcelablePackageIdentifier>() {
            @Override
            public ParcelablePackageIdentifier createFromParcel(Parcel in) {
                return new ParcelablePackageIdentifier(in);
            }

            @Override
            public ParcelablePackageIdentifier[] newArray(int size) {
                return new ParcelablePackageIdentifier[size];
            }
        };
}
